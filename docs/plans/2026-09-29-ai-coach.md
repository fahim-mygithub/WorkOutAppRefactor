**Goal:** Ship the AI coach from `docs/plans/2026-09-29-ai-coach-design.md`: lift-entry reading, missed-rep help, a floating Ask AI agent that acts on the app through typed tools (tiered apply/undo), and AI lookup of exercises missing from the library with verified demo media.

**Tech Stack:** React 18 + TS, Redux Toolkit, Vitest + Testing Library (root and, new, `functions/`), Firebase Functions v2 + firebase-admin, `@anthropic-ai/sdk`, Zod.

```ts
// functions/src/access.ts
/** Who may call the AI, and how often. Pure helpers + the Firestore gate. */
import { getApps, initializeApp } from 'firebase-admin/app';
import { getFirestore } from 'firebase-admin/firestore';

export function isAllowed(email: string | undefined, list: string): boolean {
  if (!email) return false;
  const allowed = list.split(',').map((e) => e.trim().toLowerCase()).filter(Boolean);
  return allowed.includes(email.toLowerCase());
}

export interface Usage { day: string; count: number }

export function nextUsage(
  prev: Usage | undefined,
  today: string,
  limit: number,
): { ok: true; usage: Usage } | { ok: false } {
  const count = prev?.day === today ? prev.count : 0;
  if (count >= limit) return { ok: false };
  return { ok: true, usage: { day: today, count: count + 1 } };
}

/** Counts one request for `uid`; false once today's limit is reached. */
export async function takeDailySlot(uid: string, limit: number): Promise<boolean> {
  if (!getApps().length) initializeApp();
  const ref = getFirestore().collection('aiUsage').doc(uid);
  const today = new Date().toISOString().slice(0, 10);
  return getFirestore().runTransaction(async (tx) => {
    const snap = await tx.get(ref);
    const next = nextUsage(snap.data() as Usage | undefined, today, limit);
    if (!next.ok) return false;
    tx.set(ref, next.usage);
    return true;
  });
}
```

```ts
function codeOf(error: unknown): AiErrorCode {
  const code = (error as { code?: string } | null)?.code ?? '';
  if (code.endsWith('permission-denied') || code.endsWith('unauthenticated')) return 'not-allowed';
  if (code.endsWith('resource-exhausted')) return 'limit';
  if (code.endsWith('failed-precondition')) return 'unconfigured';
  return 'failed';
}

/** One call to the `ai` callable; throws AiBackendError with a code. */
export async function callAi<T>(payload: AiCallablePayload, deps?: AiClientDeps): Promise<T> {
  if (isOffline()) throw new AiBackendError('AI is unavailable while offline.', 'offline');
  const callable = getCallable(deps);
  if (!callable) throw new AiBackendError('AI backend is not configured.', 'unconfigured');
  try {
    const { data } = await callable(payload);
    return data as T;
  } catch (error) {
    throw new AiBackendError('AI request failed.', codeOf(error), error);
  }
}
```

```ts
// src/lib/aiLoadCheck.ts
/**
 * The engine's gate on AI-proposed loads: within ±30% of the reference (what
 * the lifter is or was about to lift) and on the equipment's step. A small
 * miss (off-step) snaps; anything else is rejected with a reason.
 */
import { roundLoad } from './trackedLiftProgression';
import type { WeightUnit } from '../types/trackedLifts';

export const MAX_CHANGE = 0.3;

export type LoadCheck =
  | { ok: true; value: number; snapped: boolean }
  | { ok: false; reason: string };

export function checkProposedLoad(
  proposed: number,
  reference: number | null,
  unit: WeightUnit,
  step?: number,
): LoadCheck {
  if (!(proposed > 0)) return { ok: false, reason: 'Load must be above zero.' };
  if (reference != null && reference > 0 && Math.abs(proposed / reference - 1) > MAX_CHANGE) {
    return { ok: false, reason: `${proposed} ${unit} is too far from ${reference} ${unit}.` };
  }
  const value = roundLoad(proposed, unit, step);
  return { ok: true, value, snapped: value !== proposed };
}
```

```ts
// src/ai/liftFields.ts
/** The AI's flat load/target fields (see functions/src/tools.ts) as app types. */
import type { TrackedLoad, TrackedTarget, WeightUnit } from '../types/trackedLifts';

export interface FlatLift {
  loadKind: 'weight' | 'bodyweight' | 'level';
  weight?: number;
  unit?: WeightUnit;
  level?: string;
  targetKind: 'reps' | 'repMax' | 'time' | 'none';
  reps?: number;
  repsMax?: number;
  seconds?: number;
  tempo?: string;
}

export function fromFlatLift(f: FlatLift): { load: TrackedLoad; target: TrackedTarget } | null {
  const unit: WeightUnit = f.unit ?? 'lb';
  let load: TrackedLoad;
  if (f.loadKind === 'weight') {
    if (!(f.weight && f.weight > 0)) return null;
    load = { kind: 'weight', value: f.weight, unit };
  } else if (f.loadKind === 'bodyweight') {
    load = f.weight && f.weight > 0 ? { kind: 'bodyweight', plus: { value: f.weight, unit } } : { kind: 'bodyweight' };
  } else {
    const label = f.level?.trim();
    if (!label) return null;
    load = { kind: 'level', label };
  }

  let target: TrackedTarget;
  if (f.targetKind === 'reps') {
    if (!f.reps) return null;
    target = f.repsMax && f.repsMax > f.reps ? { kind: 'reps', min: f.reps, max: f.repsMax } : { kind: 'reps', min: f.reps };
  } else if (f.targetKind === 'repMax') {
    if (!f.reps) return null;
    target = { kind: 'repMax', reps: f.reps };
  } else if (f.targetKind === 'time') {
    if (!f.seconds) return null;
    const tempo = f.tempo?.trim();
    target = tempo ? { kind: 'time', seconds: f.seconds, tempo } : { kind: 'time', seconds: f.seconds };
  } else {
    target = { kind: 'none' };
  }
  return { load, target };
}
```

```ts
// src/ai/applyTool.ts
/**
 * Turns one AI tool call into Redux actions, tiered per the design:
 *   auto     — only today's workout, just described → apply now, Undo restores
 *   confirm  — saved data or future numbers → Apply/Reject card
 *   rejected — unknown ids, bad shapes, loads the engine refuses
 * Pure: reads a snapshot, returns actions; the caller dispatches.
 */
import type { UnknownAction } from '@reduxjs/toolkit';
import type { ActiveWorkout, Exercise, WorkoutExercise, WorkoutSet } from '../types/exercise';
import type { TrackedLift } from '../types/trackedLifts';
import { updateExercise, replaceExerciseMovement } from '../store/slices/workoutSlice';
import { liftAdded, liftRemoved, liftUpdated } from '../store/slices/trackedLiftsSlice';
import { checkProposedLoad } from '../lib/aiLoadCheck';
import { formatLoad, formatTarget } from '../lib/trackedLifts';
import { fromFlatLift, type FlatLift } from './liftFields';
import type { AiToolCall } from './types';

export interface PlanContext {
  activeWorkout: ActiveWorkout | null;
  trackedLifts: TrackedLift[];
  library: Exercise[];
}

export type ToolPlan =
  | { kind: 'auto'; id: string; summary: string; apply: UnknownAction[]; undo: UnknownAction[] }
  | { kind: 'confirm'; id: string; summary: string; detail?: string; apply: UnknownAction[] }
  | { kind: 'rejected'; id: string; reason: string };

const unitOf = (set?: WorkoutSet): 'lb' | 'kg' => (set?.unit === 'kg' ? 'kg' : 'lb');
const str = (v: unknown): string | undefined => (typeof v === 'string' && v.trim() ? v : undefined);
const num = (v: unknown): number | undefined => (typeof v === 'number' && Number.isFinite(v) ? v : undefined);

function findExercise(ctx: PlanContext, id: unknown): WorkoutExercise | undefined {
  return ctx.activeWorkout?.exercises.find((e) => e.id === id);
}

/** The load the lifter is working with: next undone set, else the last set. */
function referenceLoad(ex: WorkoutExercise): number | null {
  const next = ex.sets.find((s) => !s.completed) ?? ex.sets[ex.sets.length - 1];
  return next?.weight ?? null;
}

function stepFor(ctx: PlanContext, ex: WorkoutExercise): number | undefined {
  return ex.tracked ? ctx.trackedLifts.find((l) => l.id === ex.tracked!.liftId)?.step : undefined;
}

function libraryMatch(ctx: PlanContext, name: string): Exercise | undefined {
  const key = name.trim().toLowerCase();
  return ctx.library.find((e) => e.name.toLowerCase() === key);
}

function setsAction(ex: WorkoutExercise, sets: WorkoutSet[]): UnknownAction {
  return updateExercise({ exerciseId: ex.id, sets, restTime: ex.restTime ?? 120 });
}

export function planToolCall(call: AiToolCall, ctx: PlanContext): ToolPlan {
  const { id, input } = call;
  const i = input as Record<string, unknown>;
  const reject = (reason: string): ToolPlan => ({ kind: 'rejected', id, reason });

  switch (call.name) {
    case 'adjustSet': {
      const ex = findExercise(ctx, i.exerciseId);
      const from = num(i.fromSetIndex);
      if (!ex || from === undefined) return reject('That exercise is not in this workout.');
      const unit = unitOf(ex.sets[0]);
      let weight: number | undefined;
      if (i.weight !== undefined) {
        const check = checkProposedLoad(num(i.weight) ?? 0, referenceLoad(ex), unit, stepFor(ctx, ex));
        if (!check.ok) return reject(check.reason);
        weight = check.value;
      }
      const reps = num(i.reps);
      const sets = ex.sets.map((s, idx) =>
        idx >= from && !s.completed
          ? { ...s, ...(weight !== undefined ? { weight } : {}), ...(reps !== undefined ? { reps, repMin: undefined, repMax: undefined } : {}) }
          : s,
      );
      const what = [weight !== undefined && `${weight} ${unit}`, reps !== undefined && `${reps} reps`].filter(Boolean).join(' × ');
      return {
        kind: 'auto', id,
        summary: `${ex.customTitle || ex.exercise.name}: remaining sets at ${what}`,
        apply: [setsAction(ex, sets)],
        undo: [setsAction(ex, ex.sets)],
      };
    }

    case 'logSet': {
      const ex = findExercise(ctx, i.exerciseId);
      const reps = num(i.reps);
      if (!ex || reps === undefined) return reject('That exercise is not in this workout.');
      const idx = num(i.setIndex) ?? ex.sets.findIndex((s) => !s.completed);
      if (idx < 0 || idx >= ex.sets.length) return reject('No open set to log.');
      const weight = num(i.weight);
      const sets = ex.sets.map((s, k) =>
        k === idx ? { ...s, reps, ...(weight !== undefined ? { weight } : {}), rir: num(i.rir), completed: true } : s,
      );
      return {
        kind: 'auto', id,
        summary: `Logged set ${idx + 1}: ${reps} reps${weight !== undefined ? ` at ${weight}` : ''}`,
        apply: [setsAction(ex, sets)],
        undo: [setsAction(ex, ex.sets)],
      };
    }

    case 'swapExercise': {
      const ex = findExercise(ctx, i.exerciseId);
      const name = str(i.replacementExerciseName);
      if (!ex || !name) return reject('That exercise is not in this workout.');
      const replacement = libraryMatch(ctx, name);
      if (!replacement) return reject(`"${name}" is not in the exercise library.`);
      const apply: UnknownAction[] = [replaceExerciseMovement({ exerciseId: ex.id, exercise: replacement })];
      const weight = num(i.weight);
      if (weight !== undefined) {
        apply.push(setsAction(ex, ex.sets.map((s) => (s.completed ? s : { ...s, weight }))));
      }
      if (i.scope === 'ongoing' && ex.tracked) {
        apply.push(liftUpdated({ id: ex.tracked.liftId, name: replacement.name }));
      }
      return {
        kind: 'confirm', id,
        summary: `Swap ${ex.exercise.name} for ${replacement.name}${i.scope === 'ongoing' ? ' from now on' : ' today'}`,
        detail: str(i.reason),
        apply,
      };
    }

    case 'updateBenchmark': {
      const lift = ctx.trackedLifts.find((l) => l.id === i.liftId);
      const parsed = fromFlatLift(i as unknown as FlatLift);
      if (!lift || !parsed) return reject('That tracked lift or benchmark is not valid.');
      return {
        kind: 'confirm', id,
        summary: `${lift.name} benchmark → ${[formatLoad(parsed.load), formatTarget(parsed.target)].filter(Boolean).join(', ')}`,
        detail: str(i.reason),
        apply: [liftUpdated({ id: lift.id, ...parsed })],
      };
    }

    case 'addTrackedLift': {
      const name = str(i.name);
      const category = str(i.category);
      const parsed = fromFlatLift(i as unknown as FlatLift);
      if (!name || !category || !parsed) return reject('That lift is missing a name, category or benchmark.');
      return {
        kind: 'confirm', id,
        summary: `Track ${name} (${category}): ${[formatLoad(parsed.load), formatTarget(parsed.target)].filter(Boolean).join(', ')}`,
        apply: [liftAdded({ name, category, ...parsed, sets: num(i.sets), progression: i.progression === true })],
      };
    }

    case 'removeTrackedLift': {
      const lift = ctx.trackedLifts.find((l) => l.id === i.liftId);
      if (!lift) return reject('That tracked lift does not exist.');
      return { kind: 'confirm', id, summary: `Stop tracking ${lift.name}`, detail: str(i.reason), apply: [liftRemoved(lift.id)] };
    }

    default:
      return reject(`Unknown tool ${String(call.name)}.`);
  }
}
```

```ts
// functions/src/media.ts
/** Checks a candidate demo clip the way the app will load it: https, reachable
 *  with the app as Referer (catches hotlink blocks), a real media type, sane size. */
const TYPES = ['video/mp4', 'video/webm', 'image/gif'];
export const MAX_BYTES = 25_000_000;

export type MediaCheck = { ok: true; type: string } | { ok: false; reason: string };

export async function verifyMedia(url: string, referer: string, fetchImpl: typeof fetch = fetch): Promise<MediaCheck> {
  if (!url.startsWith('https://')) return { ok: false, reason: 'not https' };
  const headers = { Referer: referer, 'User-Agent': 'Mozilla/5.0 (WorkoutApp media check)' };
  let r = await fetchImpl(url, { method: 'HEAD', headers, redirect: 'follow' }).catch(() => null);
  if (!r || r.status === 405 || r.status === 501) {
    r = await fetchImpl(url, { method: 'GET', headers: { ...headers, Range: 'bytes=0-0' }, redirect: 'follow' }).catch(() => null);
  }
  if (!r || !r.ok) return { ok: false, reason: `status ${r?.status ?? 'error'}` };
  const type = (r.headers.get('content-type') ?? '').split(';')[0].trim().toLowerCase();
  if (!TYPES.includes(type)) return { ok: false, reason: `type ${type || 'unknown'}` };
  const range = r.headers.get('content-range')?.split('/')[1];
  const size = Number(range ?? r.headers.get('content-length') ?? 0);
  if (size > MAX_BYTES) return { ok: false, reason: 'too large' };
  return { ok: true, type };
}
```

Design points:
- Tools passed to the model: Anthropic's server web search `{ type: 'web_search_20250305', name: 'web_search', max_uses: 5 }` plus one client tool `submitExercise` (Zod → JSON via the existing converter). The prompt tells the model to finish by calling `submitExercise` exactly once.
- Loop: while `stop_reason === 'pause_turn'`, resend with the assistant content appended (max 3 rounds). Then read the `submitExercise` tool_use input; validate with `exerciseLookupSchema`.
- **Prompt-injection guard:** only `submitExercise` input is read; everything else from the model is ignored. No other tool is offered in this action.
- Verify each `mediaCandidates` URL with `verifyMedia(url, APP_ORIGIN)` (`APP_ORIGIN = defineString('APP_ORIGIN', { default: 'https://fahim-mygithub.github.io/' })`); return the first 3 that pass as `media`, plus `rejectedMedia` count.
- `aliasOf`: a standard name the exercise is commonly listed under ("Lying Triceps Extension" for "skull crusher"); the client checks it against the library.

Schema:
